def read(f):return json.loads(Path(f).read_text())
def save(f,v):
    f=Path(f);tmp=Path(str(f)+'.partial')
    with tmp.open('w') as o:json.dump(v,o,ensure_ascii=False,separators=(',',':'));o.write('\n');o.flush();os.fsync(o.fileno())
    tmp.replace(f)
def sha(f):return hashlib.sha256(Path(f).read_bytes()).hexdigest()
def stopped(folder):
    ids=[]
    for name in ['summary.json','audit.json']:
        if (folder/name).exists():ids.append(read(folder/name)['pid'])
    if (folder/'worker_generations.json').exists():
        for g in read(folder/'worker_generations.json'):ids+=[g['parent']]+g['workers']
    assert all(not Path('/proc',str(pid)).exists() for pid in ids);return ids
def manifest(folder):
    stopped(folder);rr=[dict(path=str(f.relative_to(ROOT)),bytes=f.stat().st_size,sha256=sha(f)) for f in sorted(folder.rglob('*')) if f.is_file() and f.name!='local_manifest.json']
    save(folder/'local_manifest.json',dict(state='LOCAL_ALL_SHA_PASS_REMOTE_PENDING',files=rr))
    assert all(sha(ROOT/r['path'])==r['sha256'] for r in rr)
def call(action,mode,name,seconds):
    assert seconds>0
    with (OUT/(name+'_'+action+'.log')).open('xb') as log:
        c=subprocess.Popen(['taskset','-c','0',str(ROOT/'run_stage.sh'),str(HERE/'t493_runtime.py'),action,mode,name],cwd=ROOT,stdout=log,stderr=subprocess.STDOUT,start_new_session=True)
        with (OUT/'subprocesses.jsonl').open('a') as f:f.write(json.dumps(dict(pid=c.pid,pgid=c.pid,action=action,mode=mode,child=name,epoch=time.time()))+'\n');f.flush();os.fsync(f.fileno())
        try:
            c.wait(timeout=seconds)
            assert c.returncode==0,('Child failed',name,action,c.returncode)
        except BaseException:
            try:os.killpg(c.pid,signal.SIGTERM)
            except ProcessLookupError:pass
            try:c.wait(timeout=3)
            except subprocess.TimeoutExpired:
                try:os.killpg(c.pid,signal.SIGKILL)
                except ProcessLookupError:pass
                c.wait()
            raise
def accounting():
    ss=[read(f) for f in OUT.glob('*/summary.json')]
    return dict(physics=sum(s['physics'] for s in ss),commands=read(OUT/'preflight.json')['commands']+sum(s['commands'] for s in ss),main_seconds=sum(s['elapsed_s'] for s in ss))
def choose_jobs(state,p):
    rows={r['job_id']:r for r in state['eval_rows']};by={j['id']:j for j in p['wide8_jobs']};ids=[]
    for side in ['LEFT','RIGHT']:
        failed=[r for r in rows.values() if side in r['job_id'] and not r['success']]
        if failed:
            # One contact/long-distance failure and one near-goal timeout.
            far=min(failed,key=lambda r:(not r['collision'],-float(r['remaining_m']),r['job_id']));ids.append(far['job_id'])
            rest=[r for r in failed if r['job_id']!=far['job_id']]
            if rest:ids.append(min(rest,key=lambda r:(float(r['remaining_m']),r['job_id']))['job_id'])
            else:ids.append(next(j['id'] for j in p['wide8_jobs'] if side in j['id'] and j['id']!=far['job_id']))
        else:ids += [j['id'] for j in p['wide8_jobs'] if side in j['id']][:2]
    assert len(ids)==len(set(ids))==4;return [by[x] for x in ids]
