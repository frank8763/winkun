def actor(p,path,teacher):return C.frozen(p,path).requires_grad_(False) if teacher else P.frozen(p,path)

def worker(pipe,index,p,counts):
    E.N.offline();C.ban_learning();os.sched_setaffinity(0,{index});torch.set_num_threads(1);torch.use_deterministic_algorithms(True);E.set_out(p['out'])
    folder=Path(p['out'])/'workers'/str(index);folder.mkdir(parents=True,exist_ok=True);log=(folder/'console.log').open('w');stdout,stderr=sys.stdout,sys.stderr;s=w=None
    def guard(physical=False):E.L.check_stop();assert time.time()<p['native_stop_epoch']-5 and sum(counts)<p['max_physics']
    def charge():counts[index+1]+=1;assert sum(counts)<=p['max_physics']
    try:
        with contextlib.redirect_stdout(log),contextlib.redirect_stderr(log):
            module,native,create,w,_=E.N.S.P.C.B.setup(folder/'native',E.N.science(p));original=copy.deepcopy(module.ACTIVE);E.L.install_stop_handlers();pipe.send(dict(state='READY',pid=os.getpid(),cpu=index))
            while True:
                req=pipe.recv()
                if req is None:break
                teacher=req.get('teacher',False);q=actor(p,req['model'],teacher);q.sample=req.get('seed') is not None;q.rng=None if req.get('seed') is None else np.random.default_rng(np.random.SeedSequence(req['seed']))
                j=req['job'];module.ACTIVE=copy.deepcopy(original);module.ACTIVE['circle_shift_x_m']=j.get('circle_shift_x_m',original['circle_shift_x_m'])
                before=counts[index+1];s=E.L.Session(j,native,create,w,guard,charge,q);E.L.install_stop_handlers()
                if not teacher:s.hidden=torch.zeros((1,1,64),dtype=torch.float64)
                if req.get('cancel_probe'):
                    def cancel(physical=False):
                        if physical:raise E.L.StageStop('T446 zero-native cancellation')
                    s.guard=cancel
                    try:s.step();raise AssertionError('Cancellation missing')
                    except E.L.StageStop:pass
                    assert not s.rows and s.steps==0;s.close();s=None;pipe.send(dict(state='CANCEL_PASS',physics=0));continue
                while not s.done():s.step(owner=0)
                a=E.payload(s);r=s.result();r.update(policy_hash=q.phash(),physics=counts[index+1]-before,array_digest=digest(a),array_hashes={k:digest(v) for k,v in a.items()},
                    integration_sha256=digest(a['final_native_integration']),pid=os.getpid(),cpu=index,teacher=teacher,mode='frozen_teacher_reconstruction' if teacher else 'student')
                if req.get('full_path'):
                    path=Path(req['full_path']);path.parent.mkdir(parents=True,exist_ok=True);blob=E.N.R.blob(E.N.S.packed(a));assert len(blob)<=p['full_bytes']
                    if path.exists():assert path.read_bytes()==blob
                    else:path.write_bytes(blob)
                s.close();s=None;pipe.send((a,r))
    except BaseException:
        save(folder/'failure.json',dict(error=traceback.format_exc(),physics=counts[index+1]))
        if s is not None and s.rows:
            try:E.N.write_npz(folder/'partial.npz',E.N.S.packed(E.payload(s)),p['full_bytes'])
            except BaseException:save(folder/'partial_save_error.json',dict(error=traceback.format_exc()))
        try:pipe.send(dict(state='FAILED',error=traceback.format_exc()))
        except BaseException:pass
        raise
    finally:
        if s:s.close()
        if w:w.close()
        sys.stdout,sys.stderr=stdout,stderr;log.close();pipe.close()

def synthetic(obj,p):
    q=obj[0];rng=np.random.default_rng(p['seed']);arrays=[]
    for i in range(4):
        x=(np.asarray(p['input_mean'])[None]+rng.normal(size=(9,120))*.01).astype(np.float32)
        with torch.no_grad():loc=q.sequence(torch.tensor(x,device=next(q.parameters()).device));raw=loc+loc.new_tensor(rng.normal(size=(9,2)))*.2;atom=torch.where(raw>=L.EDGE,1,torch.where(raw<=-L.EDGE,-1,0)).to(torch.int8);lp=L.density(loc,raw,atom,p['raw_sigma'])
        flags=np.zeros((9,3),np.uint8);flags[-1,1]=1
        arrays.append(dict(states=x,next_states=np.concatenate([x[1:],x[-1:]]),locations=loc.cpu().numpy(),raw_actions=raw.cpu().numpy(),atom=atom.cpu().numpy(),logp=lp.cpu().numpy(),
            rewards=rng.normal(size=9)*.01,native_counts=np.full(9,100,np.uint8),end_flags=flags,ownership=np.zeros(9,np.uint8),stochastic=np.ones(9,np.uint8)))
    return arrays

class Runtime(M.Runtime):

def audit_array(q,a,p,seed,teacher=False):
    if teacher:E.NATIVE_AUDIT.audit_array(q,a,p,seed=None);return
    q.reset_noise();q.rng=None if seed is None else np.random.default_rng(np.random.SeedSequence(seed));h=None
    with torch.no_grad():
        for i,x in enumerate(a['states']):
            q.sample=bool(a['stochastic'][i]);hb=np.zeros(64) if h is None else h[0,0].numpy().copy();cmd,loc,h,_=q.step(torch.tensor(x[None]),h)

            n=int(a['native_counts'][i]);z=E.N.K.R.GAMMA**n;f=a['end_flags'][i];d0=np.linalg.norm(a['poses'][i,:2]-a['task_goal_xy']);d1=np.linalg.norm(a['next_poses'][i,:2]-a['task_goal_xy'])
            reward=-.000002*(1-z)/(1-E.N.K.R.GAMMA)+.03*d0-z*.03*d1*(not f.any())
            if f.any():reward+=E.N.K.R.GAMMA**(n-1)*(float(f[1])-float(f[0])-.5*float(f[2]))
            assert abs(reward-a['rewards'][i])<1e-12
    assert np.array_equal(h[0,0].numpy(),a['last_hidden_after'])
